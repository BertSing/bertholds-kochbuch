(()=>{const q=document.querySelector('#friedaSearch'),cards=[...document.querySelectorAll('.historic-card')],count=document.querySelector('#friedaCount'),empty=document.querySelector('#friedaEmpty');const norm=v=>v.toLocaleLowerCase('de-DE').normalize('NFD').replace(/[\u0300-\u036f]/g,'');function update(){const t=norm(q.value.trim());let n=0;cards.forEach(c=>{const show=!t||norm(c.dataset.search).includes(t);c.hidden=!show;if(show)n++});count.textContent=`${n} ${n===1?'Rezept':'Rezepte'}`;empty.hidden=n!==0}q.addEventListener('input',update);update()})();
